	f'\brak{x}\system{L}sG\brak{s}-f\brak{0}\label{eq:laplace1d}
\end{equation}
Proof\\
Let G(s) be the laplace transform of y=f(x):\\
\begin{align}
    y\system{L}G\brak{s}
\end{align}
By definition of laplace transform,\\
\begin{align}
	G\brak{s}=\int_{0}^{\infty}f(x)e^{-sx}dx
\end{align}
integrating by parts,
\begin{align}
	G\brak{s}&=\int_0^{\infty}f\brak{x}d\brak{\frac{e^{-sx}}{-s}}\\
			 &=f\brak{x}\,\frac{e^{-sx}}{-s}\Bigg|_0^{\infty}+\int_0^{\infty}f'\brak{x}\,\frac{e^{-sx}}{s}\\
			 &=\brak{\frac{f\brak{\infty}e^{-\infty}-f\brak{0}}{-s}}+\frac{1}{s}\,\mathcal{L}\{f'\brak{x}\}\\
			 sG\brak{s}&=f\brak{0}+\mathcal{L}\{f'(x)\}\\
			 f'\brak{x}&\system{L}sG\brak{s}-f\brak{0}			 		
\end{align}
