Laplace transforms of some common functions:\label{eq:laplacecommon}

\item
\begin{align}
\cos\brak{at}\system{L}\frac{s}{s^2+a^2}\label{eq:laplacecos}
\end{align}

Proof:
\begin{align}
    \cos(at)&\system{L}\int_{0}^{\infty}\cos\brak{at}\,e^{-st}\,dt=I\\
                          I&=\int_{0}^{\infty}\frac{d\brak{\sin\brak{at}}}{a}\,e^{-st}\\
                           &=\frac{\sin\brak{at}}{a}\,e^{-st}\Bigg|_0^\infty-\frac{1}{a}\int_{0}^{\infty}\sin(at)e^{-st}\brak{-s}dt\\
                           &=0+\frac{s}{a}\int_0^\infty \frac{d\brak{-\cos\brak{at}}}{a}\,e^{-st}\\
                           &=-\brak{\frac{s}{a^2}}\left\{\cos\brak{at}\,e^{-st}\,\,\Bigg|_0^\infty-\int_0^\infty\cos\brak{at}\,e^{-st}\,\brak{-s}\right\}\\
                           &=-\brak{\frac{s}{a^2}}\left\{\brak{0-1}-\brak{-s}I\right\}\\
                           &=-\brak{\frac{s}{a^2}}\left\{-1+sI\right\}\\
                          I&=\frac{s}{a^2}-\frac{s^2}{a^2}\brak{I}\\
       \brak{\frac{a^2+s^2}{a^2}}I&=\frac{s}{a^2}\\
                          I&=\frac{s}{s^2+a^2}
\end{align}
\item
\begin{align}
\sin\brak{at}&\system{L}\frac{a}{s^2+a^2}\label{eq:laplacesin}
\end{align}
Proof:
\begin{align}
    \sin\brak{at}&\system{L}\int_{0}^{\infty}\sin\brak{at}\,e^{-st}\,dt=I\\
                          I&=\int_{0}^{\infty}\frac{d\brak{-\cos\brak{at}}}{a}\,e^{-st}\\
                           &=\frac{-\cos\brak{at}}{a}\,e^{-st}\Bigg|_0^\infty-\frac{1}{a}\int_{0}^{\infty}-\cos(at)e^{-st}(-s)dt\\
                           &=\frac{1}{a}-\frac{s}{a}\int_0^\infty \frac{d\brak{\sin\brak{at}}}{a}\,e^{-st}\\
                           &=\frac{1}{a}-(\frac{s}{a^2})\left\{\sin(at)\,e^{-st}\,\,\Bigg|_0^\infty-\int_0^\infty\sin\brak{at}\,e^{-st}\,\brak{-s}\right\}\\
                           &=-\brak{\frac{s}{a^2}}\left\{\brak{0-0}-\brak{-s}I\right\}\\
                           &=-\brak{\frac{s}{a^2}}\left\{0+sI\right\}\\
                          I&=\frac{a}{a^2}-\frac{s^2}{a^2}\brak{I}\\
       \brak{\frac{a^2+s^2}{a^2}}I&=\frac{a}{a^2}\\
                          I&=\frac{a}{s^2+a^2}
\end{align}
\item
\begin{align}
\sinh\brak{at}\system{L}\frac{a}{s^2-a^2}\label{eq:laplacesinh}
\end{align}
Proof:
\begin{align}
\sinh\brak{at}&\system{L}\int_0^\infty \frac{e^{at}-e^{-at}}{2}\,e^{-st} dt\\
                            &=\frac{1}{2}\int_0^\infty e^{\brak{a-s}\brak{t}} dt -\frac{1}{2}\int_0^\infty e^{\brak{-a-s}\brak{t}}dt\\
                            &=\frac{1}{2}\left\{\frac{e^{\brak{a-s}\brak{t}}}{a-s}\right\}\Bigg|_0^\infty-\frac{1}{2}\left\{\frac{e^{\brak{-a-s}\brak{t}}}{-a-s}\right\}\Bigg|_0^\infty\tag{assume s$>$a for integral to be finite}\\
                            &=\frac{1}{2}\brak{\frac{1}{s-a}-\frac{1}{s+a}}\\
                            &=\frac{1}{2}\brak{\frac{2a}{s^2-a^2}}\\
                           I&=\frac{a}{s^2-a^2}
\end{align}
\item
\begin{align}
\cosh\brak{at}\system{L}\frac{s}{s^2-a^2}\label{eq:laplacecosh}
\end{align}
Proof:
\begin{align}
\cosh\brak{at}&\system{L}\int_0^\infty \frac{e^{at}+e^{-at}}{2}\,e^{-st} dt\\
                            &=\frac{1}{2}\int_0^\infty e^{\brak{a-s}\brak{t}} dt +\frac{1}{2}\int_0^\infty e^{\brak{-a-s}\brak{t}}dt\\
                            &=\frac{1}{2}\left\{\frac{e^{\brak{a-s}\brak{t}}}{a-s}\right\}\Bigg|_0^\infty+\frac{1}{2}\left\{\frac{e^{\brak{-a-s}\brak{t}}}{-a-s}\right\}\Bigg|_0^\infty\tag{assume s$>$a for integral to be finite}\\
                            &=\frac{1}{2}\brak{\frac{1}{s-a}+\frac{1}{s+a}}\\
                            &=\frac{1}{2}\brak{\frac{2s}{s^2-a^2}}\\
                           I&=\frac{s}{s^2-a^2}
\end{align}

\item
\begin{equation}
	f'\brak{x}\system{L}sG\brak{s}-f\brak{0}\label{eq:laplace1d}
\end{equation}
Proof\\
Let G(s) be the laplace transform of y=f(x):\\
\begin{align}
    y\system{L}G\brak{s}
\end{align}
By definition of laplace transform,\\
\begin{align}
	G\brak{s}=\int_{0}^{\infty}f(x)e^{-sx}dx
\end{align}
integrating by parts,
\begin{align}
	G\brak{s}&=\int_0^{\infty}f\brak{x}d\brak{\frac{e^{-sx}}{-s}}\\
			 &=f\brak{x}\,\frac{e^{-sx}}{-s}\Bigg|_0^{\infty}+\int_0^{\infty}f'\brak{x}\,\frac{e^{-sx}}{s}\\
			 &=\brak{\frac{f\brak{\infty}e^{-\infty}-f\brak{0}}{-s}}+\frac{1}{s}\,\mathcal{L}\{f'\brak{x}\}\\
			 sG\brak{s}&=f\brak{0}+\mathcal{L}\{f'(x)\}\\
			 f'\brak{x}&\system{L}sG\brak{s}-f\brak{0}			 		
\end{align}
